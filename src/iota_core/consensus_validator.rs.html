25/// Allows verifying the validity of transactions
26#[derive(Clone)]
27pub struct IotaTxValidator {
28    epoch_store: Arc<AuthorityPerEpochStore>,
29    checkpoint_service: Arc<dyn CheckpointServiceNotify + Send + Sync>,
30    _transaction_manager: Arc<TransactionManager>,
31    metrics: Arc<IotaTxValidatorMetrics>,
32}
33
34impl IotaTxValidator {
35    pub fn new(
36        epoch_store: Arc<AuthorityPerEpochStore>,
37        checkpoint_service: Arc<dyn CheckpointServiceNotify + Send + Sync>,
38        transaction_manager: Arc<TransactionManager>,
39        metrics: Arc<IotaTxValidatorMetrics>,
40    ) -> Self {
41        info!(
42            "IotaTxValidator constructed for epoch {}",
43            epoch_store.epoch()
44        );
45        Self {
46            epoch_store,
47            checkpoint_service,
48            _transaction_manager: transaction_manager,
49            metrics,
50        }
51    }

53    #[instrument(level = "trace", skip_all)]
54    fn validate_transactions(&self, txs: Vec<ConsensusTransactionKind>) -> Result<(), IotaError> {
55        let mut cert_batch = Vec::new();
56        let mut ckpt_messages = Vec::new();
57        let mut ckpt_batch = Vec::new();
58        let mut authority_cap_batch = Vec::new();
59        let mut user_tx_v1_count: u64 = 0;
60
61        for tx in txs.iter() {
62            match tx {
63                ConsensusTransactionKind::CertifiedTransaction(certificate) => {
64                    cert_batch.push(certificate.as_ref());
65                }
66                ConsensusTransactionKind::CheckpointSignature(signature) => {
67                    ckpt_messages.push(signature.as_ref());
68                    ckpt_batch.push(&signature.summary);
69                }
70                ConsensusTransactionKind::RandomnessDkgMessage(_, bytes) => {
71                    if bytes.len() > dkg_v1::DKG_MESSAGES_MAX_SIZE {
72                        warn!("batch verification error: DKG Message too large");
73                        return Err(IotaError::InvalidDkgMessageSize);
74                    }
75                }
76                ConsensusTransactionKind::RandomnessDkgConfirmation(_, bytes) => {
77                    if bytes.len() > dkg_v1::DKG_MESSAGES_MAX_SIZE {
78                        warn!("batch verification error: DKG Confirmation too large");
79                        return Err(IotaError::InvalidDkgMessageSize);
80                    }
81                }
82                ConsensusTransactionKind::SignedCapabilityNotificationV1(signed_cap) => {
83                    authority_cap_batch.push(signed_cap);
84                }
85
86                ConsensusTransactionKind::MisbehaviorReport(_) => {
87                    if !self
88                        .epoch_store
89                        .protocol_config()
90                        .calculate_validator_scores()
91                    {
92                        return Err(IotaError::UnsupportedFeature {
93                            error: "MisbehaviorReport not supported at current protocol version"
94                                .into(),
95                        });
96                    }
97                }
98                #[allow(deprecated)]
99                ConsensusTransactionKind::NewJWKFetchedDeprecated => {
100                    return Err(IotaError::UnsupportedFeature {
101                        error: "NewJWKFetched (zkLogin) is deprecated and not supported".into(),
102                    });
103                }
104
105                ConsensusTransactionKind::UserTransactionV1(transaction) => {
106                    if !self.epoch_store.protocol_config().enable_pcool_flow() {
107                        return Err(IotaError::UnsupportedFeature {
108                            error: "UserTransactionV1 not supported at current protocol version"
109                                .into(),
110                        });
111                    }
112                    // TODO: Batch signature verification for UserTransactionV1.
113                    //  For now verify individually, but this should be batched for performance
114                    //  similar to how certificates are batch-verified above.
115                    self.epoch_store
116                        .signature_verifier
117                        .verify_tx(transaction.data())
118                        .tap_err(|e| {
119                            warn!("UserTransactionV1 signature verification failed: {}", e)
120                        })?;
121                    user_tx_v1_count += 1;
122                }
123
124                ConsensusTransactionKind::EndOfPublish(_)
125                | ConsensusTransactionKind::CapabilityNotificationV1(_) => {}
126
127                ConsensusTransactionKind::OverloadNotificationV1(authority_name, _, percentage) => {
128                    if !self.epoch_store.protocol_config().enable_pcool_flow() {
129                        return Err(IotaError::UnsupportedFeature {
130                            error:
131                                "OverloadNotificationV1 not supported at current protocol version"
132                                    .into(),
133                        });
134                    }
135                    if *percentage > 100 {
136                        return Err(IotaError::HandleConsensusTransactionFailure(format!(
137                            "OverloadNotificationV1 with invalid percentage {percentage} from \
138                                authority {}",
139                            authority_name.concise(),
140                        )));
141                    }
142                }
143            }
144        }
145
146        // verify the certificate signatures as a batch
147        let cert_count = cert_batch.len();
148        let ckpt_count = ckpt_batch.len();
149        let authority_cap_count = authority_cap_batch.len();
150
151        self.epoch_store
152            .signature_verifier
153            .verify_certs_and_checkpoints(cert_batch, ckpt_batch, authority_cap_batch)
154            .tap_err(|e| warn!("batch verification error: {}", e))?;
155
156        // All checkpoint sigs have been verified, forward them to the checkpoint
157        // service
158        for ckpt in ckpt_messages {
159            self.checkpoint_service
160                .notify_checkpoint_signature(&self.epoch_store, ckpt)?;
161        }
162
163        self.metrics
164            .certificate_signatures_verified
165            .inc_by(cert_count as u64);
166        self.metrics
167            .checkpoint_signatures_verified
168            .inc_by(ckpt_count as u64);
169        self.metrics
170            .authority_capabilities_verified
171            .inc_by(authority_cap_count as u64);
172        self.metrics
173            .user_transaction_signatures_verified
174            .inc_by(user_tx_v1_count);
175        Ok(())
176
177        // todo - we should un-comment line below once we have a way to revert
178        // those transactions at the end of epoch all certificates had
179        // valid signatures, schedule them for execution prior to sequencing
180        // which is unnecessary for owned object transactions.
181        // It is unnecessary to write to pending_certificates table because the
182        // certs will be written via consensus output.
183        // self.transaction_manager
184        //     .enqueue_certificates(owned_tx_certs, &self.epoch_store)
185        //     .wrap_err("Failed to schedule certificates for execution")
186    }
187}

189fn tx_from_bytes(tx: &[u8]) -> Result<ConsensusTransaction, eyre::Report> {
190    bcs::from_bytes::<ConsensusTransaction>(tx)
191        .wrap_err("Malformed transaction (failed to deserialize)")
192}
193
194impl starfish_core::TransactionVerifier for IotaTxValidator {
195    #[instrument(level = "trace", skip_all)]
196    fn verify_batch(
197        &self,
198        batch: &[&[u8]],
199    ) -> core::result::Result<(), starfish_core::ValidationError> {
200        let _scope = monitored_scope("ValidateBatch");
201
202        let txs = batch
203            .iter()
204            .map(|tx| {
205                tx_from_bytes(tx)
206                    .map(|tx| tx.kind)
207                    .map_err(|e| starfish_core::ValidationError::InvalidTransaction(e.to_string()))
208            })
209            .collect::<core::result::Result<Vec<_>, _>>()?;
210
211        self.validate_transactions(txs)
212            .map_err(|e| starfish_core::ValidationError::InvalidTransaction(e.to_string()))
213    }
214}

216pub struct IotaTxValidatorMetrics {
217    certificate_signatures_verified: IntCounter,
218    checkpoint_signatures_verified: IntCounter,
219    authority_capabilities_verified: IntCounter,
220    user_transaction_signatures_verified: IntCounter,
221}
222
223impl IotaTxValidatorMetrics {
224    pub fn new(registry: &Registry) -> Arc<Self> {
225        Arc::new(Self {
226            certificate_signatures_verified: register_int_counter_with_registry!(
227                "certificate_signatures_verified",
228                "Number of certificates verified in consensus batch verifier",
229                registry
230            )
231            .unwrap(),
232            checkpoint_signatures_verified: register_int_counter_with_registry!(
233                "checkpoint_signatures_verified",
234                "Number of checkpoint verified in consensus batch verifier",
235                registry
236            )
237            .unwrap(),
238            authority_capabilities_verified: register_int_counter_with_registry!(
239                "authority_capabilities_verified",
240                "Number of signed authority capabilities verified in consensus batch verifier",
241                registry
242            )
243            .unwrap(),
244            user_transaction_signatures_verified: register_int_counter_with_registry!(
245                "user_transaction_signatures_verified",
246                "Number of UserTransactionV1 signatures verified in consensus validator",
247                registry
248            )
249            .unwrap(),
250        })
251    }

349    /// Verifies that `validate_transactions` correctly gates every
350    /// `ConsensusTransactionKind` variant against the current protocol config's
351    /// feature flags.
352    ///
353    /// The exhaustive match forces a compile error when new variants are added,
354    /// so the developer must explicitly map each variant to its gating flag.
355    ///
356    /// NOTE: This test is primarily useful for variants that are under
357    /// development or not yet fully rolled out on all networks. Once all
358    /// feature-gated variants are enabled on every network, this test can be
359    /// ignored — its value lies in catching missing gates during the upgrade
360    /// window between code deployment and protocol activation.
361    #[sim_test]
362    async fn validate_transactions_feature_gating() {
363        use iota_protocol_config::ProtocolConfig;
364        use iota_types::crypto::{
365            AccountKeyPair, AuthorityPublicKeyBytes, deterministic_random_account_key,
366        };
367
368        use crate::test_utils::make_transfer_iota_transaction;
369
370        let (sender, sender_key): (_, AccountKeyPair) = deterministic_random_account_key();
371        let gas_object_id = ObjectId::random();
372        let gas_object = Object::with_id_owner_for_testing(gas_object_id, sender);
373
374        let network_config =
375            iota_swarm_config::network_config_builder::ConfigBuilder::new_with_temp_dir()
376                .with_objects(vec![gas_object.clone()])
377                .build();
378
379        let state = TestAuthorityBuilder::new()
380            .with_network_config(&network_config, 0)
381            .with_chain_override(Chain::Mainnet)
382            .build()
383            .await;
384
385        let rgp = state.epoch_store_for_testing().reference_gas_price();
386        let gas_ref = state.get_object(&gas_object_id).unwrap().object_ref();
387        let recipient = iota_types::crypto::get_key_pair::<AccountKeyPair>().0;
388        let signed_tx =
389            make_transfer_iota_transaction(gas_ref, recipient, None, sender, &sender_key, rgp);
390
391        let metrics = IotaTxValidatorMetrics::new(&Default::default());
392        let validator = IotaTxValidator::new(
393            state.epoch_store_for_testing().clone(),
394            Arc::new(CheckpointServiceNoop {}),
395            state.transaction_manager().clone(),
396            metrics,
397        );
398
399        let protocol_config = validator.epoch_store.protocol_config();
400        let authority = AuthorityPublicKeyBytes::default();
401
402        // Returns the feature flag value that gates a variant, or `None` if the
403        // variant is always allowed. The exhaustive match ensures this function
404        // must be updated when new variants are added to ConsensusTransactionKind.
405        #[allow(deprecated)]
406        fn is_feature_gated(
407            kind: &ConsensusTransactionKind,
408            config: &ProtocolConfig,
409        ) -> Option<bool> {
410            match kind {
411                // Always allowed (no feature flag gating).
412                ConsensusTransactionKind::CertifiedTransaction(_)
413                | ConsensusTransactionKind::CheckpointSignature(_)
414                | ConsensusTransactionKind::EndOfPublish(_)
415                | ConsensusTransactionKind::CapabilityNotificationV1(_)
416                | ConsensusTransactionKind::SignedCapabilityNotificationV1(_)
417                | ConsensusTransactionKind::RandomnessDkgMessage(_, _)
418                | ConsensusTransactionKind::RandomnessDkgConfirmation(_, _) => None,
419
420                // Gated behind `enable_pcool_flow`.
421                ConsensusTransactionKind::UserTransactionV1(_) => Some(config.enable_pcool_flow()),
422
423                // Gated behind `calculate_validator_scores`.
424                ConsensusTransactionKind::MisbehaviorReport(_) => {
425                    Some(config.calculate_validator_scores())
426                }
427
428                // Always rejected: zkLogin JWK support was never enabled on
429                // IOTA and the variant is retained only for serialization
430                // compatibility.
431                ConsensusTransactionKind::NewJWKFetchedDeprecated => Some(false),
432
433                // Gated behind `enable_pcool_flow`.
434                ConsensusTransactionKind::OverloadNotificationV1(_, _, _) => {
435                    Some(config.enable_pcool_flow())
436                }
437            }
438        }
439
440        // Variants that can be validated without signature verification setup
441        // (or that carry a valid signature, like UserTransactionV1).
442        // CertifiedTransaction, CheckpointSignature, and
443        // SignedCapabilityNotificationV1 are excluded because they require valid
444        // cryptographic signatures and would fail before reaching the feature
445        // gate check; their gating is verified by the exhaustive match above.
446        #[allow(deprecated)]
447        let testable_variants: Vec<(&str, ConsensusTransactionKind)> = vec![
448            (
449                "EndOfPublish",
450                ConsensusTransactionKind::EndOfPublish(authority),
451            ),
452            (
453                "NewJWKFetchedDeprecated",
454                ConsensusTransactionKind::NewJWKFetchedDeprecated,
455            ),
456            (
457                "CapabilityNotificationV1",
458                ConsensusTransactionKind::CapabilityNotificationV1(
459                    iota_types::messages_consensus::AuthorityCapabilitiesV1::new(
460                        authority,
461                        Chain::Mainnet,
462                        iota_types::supported_protocol_versions::SupportedProtocolVersions::SYSTEM_DEFAULT,
463                        vec![],
464                    ),
465                ),
466            ),
467            (
468                "RandomnessDkgMessage",
469                ConsensusTransactionKind::RandomnessDkgMessage(authority, vec![]),
470            ),
471            (
472                "RandomnessDkgConfirmation",
473                ConsensusTransactionKind::RandomnessDkgConfirmation(authority, vec![]),
474            ),
475            (
476                "MisbehaviorReport",
477                ConsensusTransactionKind::MisbehaviorReport(VersionedMisbehaviorReport::new_v1(
478                    authority,
479                    0,
480                    MisbehaviorObservationsV1 {
481                        faulty_blocks_provable: vec![],
482                        faulty_blocks_unprovable: vec![],
483                        missing_proposals: vec![],
484                        equivocations: vec![],
485                    },
486                )),
487            ),
488            (
489                "UserTransactionV1",
490                ConsensusTransactionKind::UserTransactionV1(Box::new(signed_tx)),
491            ),
492            (
493                "OverloadNotificationV1",
494                ConsensusTransactionKind::OverloadNotificationV1(authority, 0, 50),
495            ),
496        ];
497
498        for (name, kind) in testable_variants {
499            let gated = is_feature_gated(&kind, protocol_config);
500            let result = validator.validate_transactions(vec![kind]);
501
502            match gated {
503                Some(false) => {
504                    // Feature flag is disabled: must be rejected.
505                    assert!(
506                        matches!(&result, Err(IotaError::UnsupportedFeature { .. })),
507                        "{name}: feature flag is disabled, expected UnsupportedFeature, \
508                         got {result:?}",
509                    );
510                }
511                Some(true) | None => {
512                    // Feature flag is enabled or variant is ungated: must not
513                    // be rejected as unsupported.
514                    assert!(
515                        !matches!(&result, Err(IotaError::UnsupportedFeature { .. })),
516                        "{name}: should not be rejected as UnsupportedFeature, got {result:?}",
517                    );
518                }
519            }
520        }
521    }
